/*
ALPONA IMAGE UPLOAD PATCH
1) Upload this file beside app.js on GitHub.
2) In index.html, add:
<script src="image-upload-patch.js"></script>
immediately after:
<script src="app.js"></script>

NOTE: On GitHub Pages this stores uploaded images only in THIS browser/device.
It will not sync to other devices or customers.
*/
(function(){
let pendingImageData="";

function install(){
  const form=document.getElementById("productForm");
  if(!form) return;
  const old=form.elements["image"];
  if(!old) return;
  const label=old.closest("label");
  if(!label || label.dataset.patched==="1") return;
  label.dataset.patched="1";
  old.style.display="none";
  old.name="old_image";

  const box=document.createElement("div");
  box.innerHTML=`
    <div id="curWrap" style="display:none;margin:8px 0">
      <div style="font:12px Arial;margin-bottom:6px">Current Image</div>
      <img id="curImg" style="width:110px;height:110px;object-fit:cover;border:1px solid #ddd">
    </div>
    <input id="imgUpload" type="file" accept="image/*"
      style="display:block;width:100%;padding:10px;border:1px solid #d7c7b2;background:white;margin-top:6px">
    <div id="preWrap" style="display:none;margin-top:10px">
      <div style="font:12px Arial;margin-bottom:6px">New Image Preview</div>
      <img id="preImg" style="width:110px;height:110px;object-fit:cover;border:1px solid #ddd">
    </div>`;
  label.appendChild(box);

  document.getElementById("imgUpload").addEventListener("change",function(){
    const file=this.files&&this.files[0];
    if(!file){pendingImageData="";return;}
    if(!file.type.startsWith("image/")){alert("Choose an image file.");this.value="";return;}
    if(file.size>2*1024*1024){alert("Image is over 2 MB. Please choose a smaller one.");this.value="";return;}
    const r=new FileReader();
    r.onload=e=>{
      pendingImageData=e.target.result;
      document.getElementById("preImg").src=pendingImageData;
      document.getElementById("preWrap").style.display="block";
    };
    r.readAsDataURL(file);
  });
}

const oldOpen=window.openProductForm;
window.openProductForm=function(id){
  oldOpen(id);
  install();
  pendingImageData="";
  const up=document.getElementById("imgUpload");
  const pre=document.getElementById("preWrap");
  if(up) up.value="";
  if(pre) pre.style.display="none";
  const cw=document.getElementById("curWrap"), ci=document.getElementById("curImg");
  if(id){
    const p=products().find(x=>x.id===id);
    if(p&&p.image){ci.src=p.image;cw.style.display="block";}
  } else if(cw) cw.style.display="none";
};

window.saveProduct=function(e){
  e.preventDefault();
  const f=e.target, fd=new FormData(f), ps=products();
  const existingId=Number(fd.get("id"));
  const id=existingId||Date.now();
  const existing=existingId?ps.find(x=>x.id===existingId):null;
  const image=pendingImageData||(existing&&existing.image)||"assets/logo.jpeg";
  const p={
    id,
    name:fd.get("name"),
    category:fd.get("category"),
    price:Number(fd.get("price")),
    cost:Number(fd.get("cost")),
    stock:Number(fd.get("stock")),
    image,
    description:fd.get("description")
  };
  const ix=ps.findIndex(x=>x.id===id);
  if(ix>=0) ps[ix]=p; else ps.push(p);
  try{set("alpona_products",ps);}
  catch(err){alert("Could not save image. Try a smaller image (under 1 MB is best).");return;}
  pendingImageData="";
  closeModal("productFormModal");
  renderProducts(); renderAdmin(); toast("Product saved with image");
};

if(document.readyState==="loading") document.addEventListener("DOMContentLoaded",install);
else install();
})();
